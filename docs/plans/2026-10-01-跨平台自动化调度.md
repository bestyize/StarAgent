# 跨平台系统级自动化任务调度（automations）实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 StarAgent 构建覆盖 Android、Desktop、iOS 的跨平台系统级任务调度子系统，让托管的定时任务在应用退到后台、锁屏、甚至进程杀死后，依然能由操作系统准时唤醒执行，并在完成后通过系统通知主动送达。

**Architecture:** 
1. 平台契约层（`core:platform`）：定义 `TaskScheduler` 调度契约、`TaskSchedulers` 注册表、`WakeupTrigger` 唤醒反向句柄、`Notifier` 系统通知契约与 `Notifiers` 注册表；
2. 调度中枢层（`core:data`）：`AppState` 采用“最近优先（Earliest-First）”动态单点排期算法，只向系统预约最近的单个唤醒时间戳，并在任务完成后通过 `Notifiers` 发送预览通知；
3. 平台实现层（`core:platform` 与 `core:bootstrap`）：
   - Android：`AlarmManager.setExactAndAllowWhileIdle()` 唤醒广播 + `NotificationChannel` 系统通知栏；
   - Desktop (JVM)：JDK 原生 `ScheduledExecutorService` 守护线程池 + AWT `SystemTray` 原生托盘通知；
   - iOS：`BGTaskScheduler` 后台刷新 + `UNUserNotificationCenter` 本地推送。

**Tech Stack:** Kotlin Multiplatform 2.4.10, Coroutines 1.11.0, kotlinx-datetime 0.8.0, Android AlarmManager/NotificationManagerCompat, Java AWT SystemTray, ScheduledExecutorService, iOS BGTaskScheduler / UserNotifications.

**Spec:** [`docs/specs/2026-10-01-跨平台自动化调度设计.md`](../specs/2026-10-01-跨平台自动化调度设计.md)（Commit `e0ef03d`）

package com.starringbox.staragent.core.platform.schedule

/**
 * 跨平台系统级任务调度器。
 */
interface TaskScheduler {
    /**
     * 向操作系统预约下一次系统唤醒时刻（epoch 毫秒）。
     * [triggerAtMillis] 为 null 时表示当前没有待执行的活跃任务，清除操作系统的唤醒预约。
     */
    fun scheduleNextWakeup(triggerAtMillis: Long?)
}

private object NoopTaskScheduler : TaskScheduler {
    override fun scheduleNextWakeup(triggerAtMillis: Long?) = Unit
}

object TaskSchedulers {
    private var installed: TaskScheduler = NoopTaskScheduler

    fun install(scheduler: TaskScheduler) {
        installed = scheduler
    }

    val current: TaskScheduler get() = installed
    val available: Boolean get() = installed !== NoopTaskScheduler
}
```

package com.starringbox.staragent.core.platform.schedule

/**
 * 系统唤醒反向触发器。
 *
 * 操作系统在后台唤醒广播/线程/后台任务时，直接调用 [fire]；
 * 数据层（AppState）在构造时向其注册自己的执行函数。
 */
object WakeupTrigger {
    private var handler: (suspend () -> Unit)? = null

    fun registerHandler(action: suspend () -> Unit) {
        handler = action
    }

    suspend fun fire() {
        handler?.invoke()
    }
}
```

package com.starringbox.staragent.core.platform.schedule

import kotlinx.coroutines.runBlocking
import java.util.concurrent.Executors
import java.util.concurrent.ScheduledExecutorService
import java.util.concurrent.ScheduledFuture
import java.util.concurrent.TimeUnit

/**
 * 桌面端任务调度器：基于单线程守护线程池进行精确延迟唤醒。
 * 守护线程（Daemon Thread）保证不会阻止 JVM 进程正常退出。
 */
class DesktopTaskScheduler(
    private val executor: ScheduledExecutorService = Executors.newSingleThreadScheduledExecutor { runnable ->
        Thread(runnable).apply {
            isDaemon = true
            name = "staragent-desktop-scheduler"
        }
    }
) : TaskScheduler {

    private var currentFuture: ScheduledFuture<*>? = null

    @Synchronized
    override fun scheduleNextWakeup(triggerAtMillis: Long?) {
        currentFuture?.cancel(false)
        currentFuture = null

        if (triggerAtMillis == null) return

        val now = System.currentTimeMillis()
        val delay = maxOf(0L, triggerAtMillis - now)

        currentFuture = executor.schedule({
            runBlocking {
                WakeupTrigger.fire()
            }
        }, delay, TimeUnit.MILLISECONDS)
    }
}
```

import java.awt.Image
import java.awt.SystemTray
import java.awt.Toolkit
import java.awt.TrayIcon
import java.awt.image.BufferedImage

/**
 * 桌面端原生系统通知：基于 AWT SystemTray 弹出操作系统通知气泡。
 * 若系统无图形环境或不支持托盘（如 headless 模式），静默忽略保证不崩溃。
 */
class DesktopNotifier : Notifier {

    private val trayIcon: TrayIcon? by lazy {
        if (!SystemTray.isSupported()) return@lazy null
        runCatching {
            val tray = SystemTray.getSystemTray()
            val image: Image = BufferedImage(16, 16, BufferedImage.TYPE_INT_ARGB)
            val icon = TrayIcon(image, "StarAgent").apply {
                isImageAutoSize = true
            }
            tray.add(icon)
            icon

package com.starringbox.staragent.core.platform.schedule

import android.app.AlarmManager
import android.app.PendingIntent
import android.content.Context
import android.content.Intent
import android.os.Build

class AndroidTaskScheduler(private val context: Context) : TaskScheduler {

    private val alarmManager = context.getSystemService(Context.ALARM_SERVICE) as? AlarmManager

    override fun scheduleNextWakeup(triggerAtMillis: Long?) {
        if (alarmManager == null) return

        val intent = Intent(context, TaskAlarmReceiver::class.java)
        val pendingIntent = PendingIntent.getBroadcast(
            context,
            ALARM_REQUEST_CODE,
            intent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        if (triggerAtMillis == null) {
            alarmManager.cancel(pendingIntent)
            return
        }

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
            alarmManager.setExactAndAllowWhileIdle(
                AlarmManager.RTC_WAKEUP,
                triggerAtMillis,
                pendingIntent,
            )
        } else {
            alarmManager.setExact(
                AlarmManager.RTC_WAKEUP,
                triggerAtMillis,
                pendingIntent,
            )
        }
    }

    companion object {
        const val ALARM_REQUEST_CODE = 10086
    }
}

            .setContentText(content)
            .setStyle(NotificationCompat.BigTextStyle().bigText(content))
            .setPriority(NotificationCompat.PRIORITY_DEFAULT)
            .setAutoCancel(true)
            .build()

        runCatching {
            NotificationManagerCompat.from(context).notify(id.hashCode(), notification)
        }
    }

        const val CHANNEL_ID = "staragent_automations"
    }

package com.starringbox.staragent.core.platform.schedule

import platform.BackgroundTasks.BGAppRefreshTaskRequest
import platform.BackgroundTasks.BGTaskScheduler
import platform.Foundation.NSDate
import platform.Foundation.dateWithTimeIntervalSince1970

/**
 * iOS 端任务调度器：基于 BackgroundTasks 框架注册后台刷新请求。
 */
class IosTaskScheduler : TaskScheduler {

    override fun scheduleNextWakeup(triggerAtMillis: Long?) {
        if (triggerAtMillis == null) {
            BGTaskScheduler.sharedScheduler.cancelTaskRequestWithIdentifier(TASK_ID)
            return
        }

        val seconds = triggerAtMillis / 1000.0
        val request = BGAppRefreshTaskRequest(TASK_ID).apply {
            earliestBeginDate = NSDate.dateWithTimeIntervalSince1970(seconds)
        }

        runCatching {
            BGTaskScheduler.sharedScheduler.submitTaskRequest(request, null)
        }
    }

    companion object {
        const val TASK_ID = "com.starringbox.staragent.task.refresh"
    }
}
```

package com.starringbox.staragent.core.platform.notify

import platform.UserNotifications.UNMutableNotificationContent
import platform.UserNotifications.UNNotificationRequest
import platform.UserNotifications.UNUserNotificationCenter

/**
 * iOS 端系统通知：基于 UserNotifications 框架投递本地推送。
 */
class IosNotifier : Notifier {

    override fun notify(id: String, title: String, content: String) {
        val notificationContent = UNMutableNotificationContent().apply {
            setTitle(title)
            setBody(content)
        }

        val request = UNNotificationRequest.requestWithIdentifier(
            identifier = id,
            content = notificationContent,
            trigger = null, // null 表示立即推送
        )

        UNUserNotificationCenter.currentNotificationCenter().addNotificationRequest(request, null)
    }
}
