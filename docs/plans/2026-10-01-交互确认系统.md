# 交互确认系统（request_user_input）实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 StarAgent 中实现 `request_user_input` 交互确认工具体系，让智能体在多步循环中遇到关键分支或需要确认时能主动向用户提问、在聊天流中弹出双模交互卡片原地挂起等待用户作答，并在获得答案后继续后续执行步骤。

**Architecture:** 
1. 领域层（`core:model`）新增 `MessageBody.Question` 消息体；
2. 契约层（`core:tools`）在 `Tool` 增加 `timeoutMillis` 属性，新增 `UserInputPort` 端口与 `RequestUserInputTool` 工具（声明 5 分钟超时）；
3. 数据层（`core:data`）基于 `CompletableDeferred` 实现 `AppUserInputPort` 挂起与 `AppState.answerQuestion` 精确唤醒，`AgentLoop` 看门狗读取工具自定义超时；
4. 界面层（`feature:chat`）在 `ChatCards.kt` 实现 `QuestionCard` 双模交互卡片（快捷选项 Chip 点击即答 + 自由文本输入提交）。

**Tech Stack:** Kotlin Multiplatform 2.4.10, Kotlin Coroutines 1.11.0, Compose Multiplatform 1.11.1, kotlinx.serialization 1.11.0, kotlin-test.

**Spec:** [`docs/specs/2026-10-01-交互确认系统设计.md`](../specs/2026-10-01-交互确认系统设计.md)（Commit `69e6c8a`）

```kotlin
package com.starringbox.staragent.core.tools.builtin

import com.starringbox.staragent.core.model.ToolDefinition
import com.starringbox.staragent.core.model.ToolResult
import com.starringbox.staragent.core.tools.Tool
import com.starringbox.staragent.core.tools.ToolHost
import kotlinx.serialization.json.JsonArray
import kotlinx.serialization.json.JsonObject
import kotlinx.serialization.json.JsonPrimitive
import kotlinx.serialization.json.add
import kotlinx.serialization.json.buildJsonArray
import kotlinx.serialization.json.buildJsonObject
import kotlinx.serialization.json.put
import kotlinx.serialization.json.putJsonArray
import kotlinx.serialization.json.putJsonObject

/**
 * 向用户发起交互提问并等待选择或输入。
 *
 * 在多步执行中遇到关键分支、需要用户拍板或二次确认时使用。
 */
object RequestUserInputTool : Tool {

    override val definition = ToolDefinition(
        name = "request_user_input",
        description = "向用户提出一个需要确认或选择的问题，并挂起等待用户作答。" +
            "当且仅当遇到关键决策分支（如确认班次、确定日期）、需要用户明确授权（如清理/重置操作）、" +
            "或需要用户从多个备选中选择时使用。" +
            "不要拿无关痛痒的问题反复打扰用户。",
        inputSchema = buildJsonObject {
            put("type", "object")
            putJsonObject("properties") {
                putJsonObject("question") {
                    put("type", "string")
                    put("description", "向用户提出的问题内容，清晰表达需要用户确认什么")
                }
                putJsonObject("options") {
                    put("type", "array")
                    put("description", "供用户快速单选的选项列表（建议 2~4 个），可选")
                    putJsonObject("items") { put("type", "string") }
                }
                putJsonObject("placeholder") {
                    put("type", "string")
                    put("description", "文本输入框的占位提示文字，引导用户输入自定义内容，可选")
                }
            }
            putJsonArray("required") { add("question") }
        },
    )

    // 人机交互需要思考与打字时间，给予 5 分钟超时，豁免通用工具的 30 秒看门狗
    override val timeoutMillis: Long = 300_000L

    override suspend fun execute(args: JsonObject, host: ToolHost): ToolResult {
        val question = args.stringOrNull("question")?.trim()
        if (question.isNullOrBlank()) {
            return ToolResult(callId = "", content = "question 不能为空", isError = true)
        }

        val options = args["options"]?.let { el ->
            (el as? JsonArray)?.mapNotNull { (it as? JsonPrimitive)?.content?.trim() }
        }.orEmpty()

        val placeholder = args.stringOrNull("placeholder")?.trim()

        val answer = host.userInput.requestInput(
            agentId = host.agentId,
            question = question,
            options = options,
            placeholder = placeholder,
        )

        return ToolResult(
            callId = "",
            content = "用户给出的回答是：$answer",
        )
    }
}
```

创建 `core/data/src/commonMain/kotlin/com/starringbox/staragent/core/data/ports/UserInputPortImpl.kt`：

```kotlin
package com.starringbox.staragent.core.data.ports

import com.starringbox.staragent.core.data.AppState
import com.starringbox.staragent.core.tools.ports.UserInputPort
import kotlinx.coroutines.CompletableDeferred
import java.util.concurrent.ConcurrentHashMap

/**
 * 交互确认端口实现。
 *
 * 基于内存中的 [CompletableDeferred] 注册表实现原地挂起与按 ID 精确唤醒。
 */
class AppUserInputPort(
    private val state: AppState,
) : UserInputPort {

    private val pendingInputs = ConcurrentHashMap<String, CompletableDeferred<String>>()

    override suspend fun requestInput(
        agentId: String,
        question: String,
        options: List<String>,
        placeholder: String?,
    ): String {
        val questionId = state.nextId("q")
        val deferred = CompletableDeferred<String>()
        pendingInputs[questionId] = deferred

        try {
            state.postQuestion(
                agentId = agentId,
                questionId = questionId,
                question = question,
                options = options,
                placeholder = placeholder,
            )
            return deferred.await()
        } finally {
            pendingInputs.remove(questionId)
        }
    }

    fun submitAnswer(questionId: String, answer: String): Boolean {
        val deferred = pendingInputs.remove(questionId) ?: return false
        return deferred.complete(answer)
    }

    fun cancelQuestion(questionId: String) {
        pendingInputs.remove(questionId)?.cancel()
    }
}
```

```kotlin
    fun postQuestion(
        agentId: String,
        questionId: String,
        question: String,
        options: List<String>,
        placeholder: String?,
    ) {
        val conversation = messagesOf(agentId)
        conversation.add(
            ChatMessage(
                id = nextId("m"),
                author = Author.Agent,
                body = MessageBody.Question(
                    id = questionId,
                    question = question,
                    options = options,
                    placeholder = placeholder,
                    selectedAnswer = null,
                ),
                timeLabel = "刚刚",
            )
        )
    }

    fun answerQuestion(agentId: String, questionId: String, answer: String): Boolean {
        val completed = userInputPort.submitAnswer(questionId, answer)
        if (!completed) return false

        val conversation = messagesOf(agentId)
        val index = conversation.indexOfFirst {
            (it.body as? MessageBody.Question)?.id == questionId
        }
        if (index >= 0) {
            val old = conversation[index].body as MessageBody.Question
            conversation[index] = conversation[index].copy(
                body = old.copy(selectedAnswer = answer)
            )
        }
        persistState()
        return true
    }
```

```kotlin
@Composable
fun QuestionCard(
    question: MessageBody.Question,
    onAnswer: (String) -> Unit,
    modifier: Modifier = Modifier,
) {
    var customInput by remember { mutableStateOf("") }
    val isAnswered = question.selectedAnswer != null

    StarCard(modifier = modifier.fillMaxWidth()) {
        Column(modifier = Modifier.padding(14.dp)) {
            // 顶栏图标与标题
            Row(verticalAlignment = Alignment.CenterVertically) {
                IconTile(
                    glyph = StarIconGlyph.Sparkle, // 或问号图标
                    tint = StarPalette.Primary,
                    containerColor = StarPalette.PrimarySubtle,
                )
                Spacer(Modifier.width(8.dp))
                Text(
                    text = "智能体提问",
                    style = MaterialTheme.typography.labelMedium,
                    color = StarPalette.TextSubtle,
                )
            }

            Spacer(Modifier.height(10.dp))

            // 问题正文
            Text(
                text = question.question,
                style = MaterialTheme.typography.bodyMedium,
                color = StarPalette.TextPrimary,
            )

            Spacer(Modifier.height(12.dp))

            if (isAnswered) {
                // 已作答展示
                Row(verticalAlignment = Alignment.CenterVertically) {
                    Icon(
                        imageVector = StarIcons.Check,
                        contentDescription = null,
                        tint = StarPalette.Success,
                        modifier = Modifier.size(16.dp),
                    )
                    Spacer(Modifier.width(6.dp))
                    Text(
                        text = "已选择：${question.selectedAnswer}",
                        style = MaterialTheme.typography.bodySmall,
                        color = StarPalette.TextSubtle,
                    )
                }
            } else {
                // 待作答交互控件
                // 1. 快捷选项列表
                if (question.options.isNotEmpty()) {
                    FlowRow(
                        horizontalArrangement = Arrangement.spacedBy(8.dp),
                        verticalArrangement = Arrangement.spacedBy(8.dp),
                    ) {
                        question.options.forEach { option ->
                            StarChip(
                                label = option,
                                selected = false,
                                onClick = { onAnswer(option) },
                            )
                        }
                    }
                    Spacer(Modifier.height(12.dp))
                }

                // 2. 自定义输入框 + 提交按钮
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    verticalAlignment = Alignment.CenterVertically,
                ) {
                    StarTextField(
                        value = customInput,
                        onValueChange = { customInput = it },
                        placeholder = question.placeholder ?: "输入其他回答...",
                        modifier = Modifier.weight(1f),
                        singleLine = true,
                    )
                    Spacer(Modifier.width(8.dp))
                    StarButton(
                        text = "提交",
                        enabled = customInput.isNotBlank(),
                        onClick = {
                            if (customInput.isNotBlank()) {
                                onAnswer(customInput.trim())
                            }
                        },
                    )
                }
            }
        }
    }
}
```

```kotlin
package com.starringbox.staragent.core.data

import com.starringbox.staragent.core.llm.ChatStreamEvent
import com.starringbox.staragent.core.llm.LlmEndpoint
import com.starringbox.staragent.core.llm.LlmProtocol
import com.starringbox.staragent.core.llm.LlmTurn
import com.starringbox.staragent.core.model.MessageBody
import com.starringbox.staragent.core.model.ToolCall
import kotlinx.coroutines.async
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.flow
import kotlinx.coroutines.test.runTest
import kotlin.test.Test
import kotlin.test.assertEquals
import kotlin.test.assertNotNull
import kotlin.test.assertTrue

class QuestionEndToEndFlowTest {

    @Test
    fun agentAsksQuestionAndContinuesExecutionAfterUserAnswers() = runTest {
        var callCount = 0
        val fakeStream: (LlmEndpoint, List<LlmTurn>) -> Flow<ChatStreamEvent> = { _, history ->
            flow {
                callCount++
                if (callCount == 1) {
                    // 第 1 轮：模型请求调用 request_user_input
                    emit(
                        ChatStreamEvent.ToolCallsReady(
                            listOf(
                                ToolCall(
                                    id = "call_q1",
                                    name = "request_user_input",
                                    arguments = """{"question":"确定要执行吗？","options":["确认","放弃"]}""",
                                )
                            )
                        )
                    )
                } else {
                    // 第 2 轮：模型收到了用户的回答，给出最终回复
                    val lastTurn = history.lastOrNull()
                    val resultText = lastTurn?.toolResults?.firstOrNull()?.content ?: ""
                    emit(ChatStreamEvent.Delta("根据您的指令（$resultText），已成功处理。"))
                }
                emit(ChatStreamEvent.Completed)
            }
        }

        val state = AppState(stream = fakeStream)
        val agent = state.agents.first()

        // 异步发送消息
        val sendJob = async {
            state.send(agent.id, "帮我执行一个危险操作")
        }

        // 验证第一轮产生了 Question 卡片
        val messages = state.messagesOf(agent.id)
        val questionMsg = messages.firstOrNull { it.body is MessageBody.Question }
        assertNotNull(questionMsg)
        val qBody = questionMsg.body as MessageBody.Question
        assertEquals("确定要执行吗？", qBody.question)

        // 模拟用户点击「确认」作答
        state.answerQuestion(agent.id, qBody.id, "确认")

        // 等待整个流程执行完毕
        sendJob.await()

        // 验证第二轮模型给出的最终回复
        val lastMsg = state.messagesOf(agent.id).last()
        assertTrue(lastMsg.body is MessageBody.Text)
        val textBody = lastMsg.body as MessageBody.Text
        assertTrue(textBody.text.contains("确认"))
    }
}
```
