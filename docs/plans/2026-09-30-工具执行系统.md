# 工具执行系统 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 给 StarAgent 加上工具执行能力——模型能真的调工具、拿到结果、继续推理，让任务真会干活、执行过程可见。

**Architecture:** 定义 / 线格式 / 执行三层分开。新建 `core:tools` 模块只依赖 `core:model`，通过窄端口取能力；`core:llm` 加两家协议的渲染器与解析器；循环留在 `core:data` 抽成 `AgentLoop.kt`，用构造函数注入 LLM 函数以便用假 LLM 单测。

**Tech Stack:** Kotlin Multiplatform 2.4.10、kotlinx.serialization 1.11.0、Ktor 3.5.2、kotlinx.coroutines 1.11.0、kotlin-test。

**Spec:** [`docs/specs/2026-09-30-工具执行系统设计.md`](../specs/2026-09-30-工具执行系统设计.md)（786 行，commit `cba3b5d`）

package com.starringbox.staragent.core.model

import kotlinx.serialization.json.JsonObject

/**
 * 工具的静态描述。**纯数据，不认识任何协议。**
 *
 * 协议相关的形状（OpenAI 的 `{type:"function",function:{...}}`、Anthropic 的
 * `{name,description,input_schema}`）由 core:llm 的渲染器从它派生，
 * 这一层不出现任何一个协议字段名。
 */
data class ToolDefinition(
    val name: String,
    val description: String,
    val inputSchema: JsonObject,
)

/**
 * 模型发出的一次工具调用。
 *
 * [arguments] 是**原始 JSON 字符串**，不在这里解析。两个理由：
 * 两家协议的原始形状都是字符串；而且官方明确警告模型不总生成合法 JSON——
 * 非法 JSON 必须是一个可处理的状态，不是异常。
 */
data class ToolCall(
    val id: String,
    val name: String,
    val arguments: String,
)

/**
 * 一次工具执行的结果。有**三种表示**，因为读者有三个。
 *
 * [content] 给模型，必须完整；[logText] 给日志，可以有损、可以截断。
 * 上下文预算和事后调试是两件事，不要用同一个字符串。
 *
 * [externalContext] 为真表示这次结果含**外部来源**的内容（网页正文等），
 * 此时本轮不产生记忆——否则网页里写的"用户喜欢简洁回答"会被当成事实记下来，
 * 而且用户永远不知道。
 */
data class ToolResult(
    val callId: String,
    val content: String,
    val isError: Boolean = false,
    val externalContext: Boolean = false,
    val logText: String? = null,
)
```

package com.starringbox.staragent.core.tools

import com.starringbox.staragent.core.model.ToolDefinition
import com.starringbox.staragent.core.model.ToolResult
import kotlinx.serialization.json.JsonObject

/**
 * 一个可被模型调用的工具。
 *
 * [execute] 的 [args] 是**已经解析并校验过**的 JSON 对象——解析失败的调用
 * 不会走到这里，会在循环里直接变成一条 isError 的结果回喂给模型。
 * 所以实现里可以假定参数形状是对的（缺字段仍然要处理，见各工具自己的校验）。
 */
interface Tool {
    val definition: ToolDefinition
    suspend fun execute(args: JsonObject, host: ToolHost): ToolResult
}

`ToolRegistry.kt`：

```kotlin
package com.starringbox.staragent.core.tools

import com.starringbox.staragent.core.model.ToolDefinition

/**
 * 工具注册表。
 *
 * 一期只有内置工具，但形状跟 core:platform 的 MediaPicker / MediaSaver /
 * LocalStore 一样是"契约 + 注册表"——三期接外部工具（MCP）时不用改结构，
 * 只是多一个注册来源。
 *
 * 保留注册顺序：模型看到的工具清单顺序会影响它的选择，顺序必须是稳定的。
 */
class ToolRegistry private constructor(private val tools: List<Tool>) {

    fun find(name: String): Tool? = tools.firstOrNull { it.definition.name == name }

    fun definitions(): List<ToolDefinition> = tools.map { it.definition }

    operator fun plus(other: ToolRegistry): ToolRegistry =
        ToolRegistry(tools + other.tools.filterNot { incoming ->
            tools.any { it.definition.name == incoming.definition.name }
        })

    companion object {
        fun of(vararg tools: Tool): ToolRegistry = ToolRegistry(tools.toList())

        val Empty: ToolRegistry = ToolRegistry(emptyList())
    }
}
```

import com.starringbox.staragent.core.model.MemoryCategory
import com.starringbox.staragent.core.model.MemoryItem

    val error: String? = null,

package com.starringbox.staragent.core.llm

import com.starringbox.staragent.core.model.ToolCall
import com.starringbox.staragent.core.model.ToolDefinition
import kotlinx.serialization.json.Json
import kotlinx.serialization.json.JsonArray
import kotlinx.serialization.json.buildJsonObject
import kotlinx.serialization.json.jsonObject
import kotlinx.serialization.json.put
import kotlin.test.Test
import kotlin.test.assertEquals

private val rememberDef = ToolDefinition(
    name = "remember",
    description = "存一条记忆",
    inputSchema = buildJsonObject { put("type", "object") },
)

private fun obj(text: String) = Json.parseToJsonElement(text).jsonObject

class ToolWireTest {

    // ── 渲染 ──────────────────────────────────────

    @Test
    fun openAiRenderWrapsInFunctionObject() {
        val rendered = ToolWire.renderOpenAi(listOf(rememberDef)).toString()
        assertEquals(
            "[{\"type\":\"function\",\"function\":{\"name\":\"remember\"," +
                "\"description\":\"存一条记忆\",\"parameters\":{\"type\":\"object\"}}}]",
            rendered,
        )
    }

    @Test
    fun anthropicRenderUsesInputSchemaAndNoWrapper() {
        val rendered = ToolWire.renderAnthropic(listOf(rememberDef)).toString()
        assertEquals(
            "[{\"name\":\"remember\",\"description\":\"存一条记忆\"," +
                "\"input_schema\":{\"type\":\"object\"}}]",
            rendered,
        )
    }

    // ── OpenAI 流式累积 ────────────────────────────

    @Test
    fun openAiAccumulatesArgumentsSplitAcrossManyFrames() {
        // 参数被切成 3 片——这是最容易出偶发 bug 的地方
        val acc = OpenAiToolCallAccumulator()
        acc.accept(obj("{\"index\":0,\"id\":\"call_1\",\"function\":{\"name\":\"remember\",\"arguments\":\"{\\\"ca\"}}"))
        acc.accept(obj("{\"index\":0,\"function\":{\"arguments\":\"tegory\\\":\\\"Preference\\\",\"}}"))
        acc.accept(obj("{\"index\":0,\"function\":{\"arguments\":\"\\\"text\\\":\\\"爱吃辣\\\"}\"}}"))

        assertEquals(
            listOf(ToolCall("call_1", "remember", "{\"category\":\"Preference\",\"text\":\"爱吃辣\"}")),
            acc.finish(),
        )
    }

    @Test
    fun openAiHandlesNameAndArgumentsInDifferentFrames() {
        val acc = OpenAiToolCallAccumulator()
        acc.accept(obj("{\"index\":0,\"id\":\"call_1\",\"function\":{\"name\":\"remember\"}}"))
        acc.accept(obj("{\"index\":0,\"function\":{\"arguments\":\"{}\"}}"))
        assertEquals(listOf(ToolCall("call_1", "remember", "{}")), acc.finish())

```kotlin
package com.starringbox.staragent.core.llm

import com.starringbox.staragent.core.model.ToolCall
import com.starringbox.staragent.core.model.ToolDefinition
import kotlinx.serialization.json.JsonArray
import kotlinx.serialization.json.JsonElement
import kotlinx.serialization.json.JsonObject
import kotlinx.serialization.json.JsonPrimitive
import kotlinx.serialization.json.addJsonObject
import kotlinx.serialization.json.buildJsonArray
import kotlinx.serialization.json.put

/**
 * 工具的**渲染层**：把协议无关的 [ToolDefinition] 变成某一家 API 认识的形状。
 *
 * 这一层很薄，但它是"多服务商"成立的关键——工具的内部模型不认识协议，
 * 协议只是边缘的一个渲染器。将来加"文本协议降级"也只是多一个函数。
 */
internal object ToolWire {

    /**
     * OpenAI 兼容系：外面要包一层 `function`，schema 的键叫 `parameters`。
     *
     * `{"type":"function","function":{"name":..,"description":..,"parameters":<schema>}}`
     */
    fun renderOpenAi(defs: List<ToolDefinition>): JsonArray = buildJsonArray {
        defs.forEach { definition ->
            addJsonObject {
                put("type", "function")
                addJsonObject("function") {
                    put("name", definition.name)
                    put("description", definition.description)
                    put("parameters", definition.inputSchema)
                }
            }
        }
    }

    /**
     * Anthropic：**不包** `function`，schema 的键叫 `input_schema`。
     *
     * `{"name":..,"description":..,"input_schema":<schema>}`
     */
    fun renderAnthropic(defs: List<ToolDefinition>): JsonArray = buildJsonArray {
        defs.forEach { definition ->
            addJsonObject {
                put("name", definition.name)
                put("description", definition.description)
                put("input_schema", definition.inputSchema)
            }
        }
    }
}

/**
 * OpenAI 兼容系的工具调用累积器。
 *
 * 模型的 `arguments` 是**跨多帧拼出来的字符串**，而 `finish_reason` 单独到达，
 * 所以不能边收边解析。按 `index` 分别累积，流结束时才拼完整：
 *
 * ```
 * {"index":0,"id":"call_1","function":{"name":"remember","arguments":"{\"ca"}}
 * {"index":0,"function":{"arguments":"tegory\":\"Preference\","}}
 * {"index":0,"function":{"arguments":"\"text\":\"爱吃辣\"}"}}
 * ```
 *
 * 名字和参数可能落在不同帧里；并行调用会交错到达，所以结果**按 index 排序**，
 * 否则回喂时对不上 id。
 */
internal class OpenAiToolCallAccumulator {

    private val ids = mutableMapOf<Int, String>()
    private val names = mutableMapOf<Int, String>()
    private val args = mutableMapOf<Int, StringBuilder>()

    /** 收一帧 `delta` 对象（已经取到 `choices[0].delta` 那一层）。 */
    fun accept(delta: JsonObject) {
        val entries = delta["tool_calls"] as? JsonArray ?: return
        entries.forEach { element ->
            val entry = element as? JsonObject ?: return@forEach
            val index = entry["index"].asIntOrNull() ?: return@forEach

            entry["id"].asTextOrNull()?.let { ids[index] = it }

            val function = entry["function"] as? JsonObject
            function?.get("name").asTextOrNull()?.let { names[index] = it }
            function?.get("arguments").asTextOrNull()?.let { fragment ->
                args.getOrPut(index) { StringBuilder() }.append(fragment)
            }
        }
    }

    /**
     * 流结束时收尾。
     *
     * 没有名字的条目直接丢掉——只有 id 和参数是没法执行的，造一个哑调用
     * 只会让模型收到一条莫名其妙的错误。没有参数则给 `{}`，无参工具就是这样。
     */
    fun finish(): List<ToolCall> = ids.keys
        .sorted()
        .mapNotNull { index ->
            val name = names[index] ?: return@mapNotNull null
            val id = ids[index] ?: return@mapNotNull null
            ToolCall(
                id = id,
                name = name,
                arguments = args[index]?.toString()?.takeIf { it.isNotBlank() } ?: "{}",
            )
        }
}

/**
 * Anthropic 的工具调用累积器。
 *
 * 与 OpenAI 的两处关键差别：
 * 1. 调用不是一个数组，而是 content 块流——`content_block_start` 开一块、
 *    若干 `content_block_delta` 往里灌
 * 2. **参数不是现成的对象**。非流式时 `input` 是解析好的；但 `stream: true`
 *    下收到的是 `input_json_delta` 里的 `partial_json` **片段字符串**，要自己拼。
 *    （这一条跟非流式的形状不同，按实际 SSE 事件为准。）
 */
internal class AnthropicToolCallAccumulator {

    private val ids = mutableMapOf<Int, String>()
    private val names = mutableMapOf<Int, String>()
    private val args = mutableMapOf<Int, StringBuilder>()

    /** `content_block_start`：开一个 tool_use 块。 */
    fun start(index: Int, id: String, name: String) {
        ids[index] = id
        names[index] = name
    }

    /** `content_block_delta` 里的 `input_json_delta`：追加一段 JSON 片段。 */
    fun acceptDelta(index: Int, partialJson: String) {
        args.getOrPut(index) { StringBuilder() }.append(partialJson)
    }

    /** 同 OpenAI：按 index 排序、无参给 `{}`、缺名字丢弃。 */
    fun finish(): List<ToolCall> = ids.keys
        .sorted()
        .mapNotNull { index ->
            val name = names[index] ?: return@mapNotNull null
            val id = ids[index] ?: return@mapNotNull null
            ToolCall(
                id = id,
                name = name,
                arguments = args[index]?.toString()?.takeIf { it.isNotBlank() } ?: "{}",
            )
        }
}

/** 从 JSON 里取整数。`index` 在真响应里一定是数字，但防御性地处理其他形状。 */
internal fun JsonElement?.asIntOrNull(): Int? {
    val primitive = this as? JsonPrimitive ?: return null
    return primitive.content.toIntOrNull()
}

/**
 * OpenAI 兼容系的工具部分。
 *
 * assistant 轮：工具调用挂在消息上的 `tool_calls` 数组里，`arguments` 回成**字符串**。

 */
internal fun JsonObjectBuilder.putOpenAiToolTurns(turn: LlmTurn) {
    if (turn.toolCalls.isEmpty()) return
    putJsonArray("tool_calls") {
        turn.toolCalls.forEach { call ->
            addJsonObject {
                put("id", call.id)
                put("type", "function")
                addJsonObject("function") {
                    put("name", call.name)
                    put("arguments", call.arguments)
                }
            }
        }
    }
}

/**
 * Anthropic 的工具部分。两处跟 OpenAI 不同：
 *
 * 1. assistant 轮的工具调用是 content 里的 `tool_use` **块**，`input` 是**对象**
 * 2. 工具结果的块类型是 `tool_result`，带 `tool_use_id`（不是 `id`）
 */
internal fun JsonObjectBuilder.putAnthropicToolBlocks(turn: LlmTurn) {
    if (turn.toolCalls.isEmpty() && turn.toolResults.isEmpty()) return

    // 已经因为图片/工具结果变成块数组了就不再重复设置 content
    val existing = this["content"]
    if (existing !is JsonArray) {
        putJsonArray("content") {
            if (turn.text.isNotBlank()) {
                addJsonObject {
                    put("type", "text")
                    put("text", turn.text)
                }
            }
            turn.toolCalls.forEach { call ->
                addJsonObject {
                    put("type", "tool_use")
                    put("id", call.id)
                    put("name", call.name)
                    put("input", parseToolArguments(call.arguments))
                }
            }
            turn.toolResults.forEach { result ->
                addJsonObject {
                    put("type", "tool_result")
                    put("tool_use_id", result.callId)
                    put("content", result.content)
                    if (result.isError) put("is_error", true)
                }
            }
        }
        return
    }

    // 已经有块了（图片路径）：把工具块追加进去
    val merged = buildJsonArray {
        existing.forEach { add(it) }
        turn.toolResults.forEach { result ->
            addJsonObject {
                put("type", "tool_result")
                put("tool_use_id", result.callId)
                put("content", result.content)
                if (result.isError) put("is_error", true)
            }
        }
    }
    put("content", merged)
}

/**
 * 把模型给的参数串解析成对象，给 Anthropic 的 `input` 用。
 *
 * 解析不了就退回空对象——**这一层不报错**。真要报错也得让工具执行那一步去报，
 * 那里能把错误变成一条给模型看的结果；协议层报错只会让整轮请求失败。
 */
private fun parseToolArguments(raw: String): JsonElement =
    runCatching { Json.parseToJsonElement(raw.ifBlank { "{}" }) }
        .getOrElse { buildJsonObject { } }

```kotlin
package com.starringbox.staragent.core.data.ports

import com.starringbox.staragent.core.tools.ports.FetchResult
import com.starringbox.staragent.core.tools.ports.WebPort
import io.ktor.client.request.get
import io.ktor.client.statement.bodyAsText
import io.ktor.http.isSuccess
import com.starringbox.staragent.core.llm.httpClient

/**
 * 取网页。
 *
 * 三道自制边界（没有权限框架，所以收在实现里）：
 * 1. **只允许 https**——明文 http 会把用户看的地址暴露在链路上
 * 2. **拒绝内网地址**——否则模型可以把这台设备当跳板去探测局域网
 * 3. **长度截断**——网页正文可以极长，不截会把上下文撑爆
 *
 * 失败返回 [FetchResult.ok] = false 而**不抛异常**：抓不到是给模型看的信息，
 * 它看到"这个地址取不到"会换个地址重试。
 */
class AppWebPort : WebPort {

    override suspend fun fetch(url: String): FetchResult {
        val normalized = url.trim()
        if (!normalized.startsWith("https://")) {
            return FetchResult(false, "", "只支持 https 地址")
        }
        if (looksInternal(normalized)) {
            return FetchResult(false, "", "这个地址指向内网，不能访问")
        }

        return runCatching {
            val response = httpClient().get(normalized)
            if (!response.status.isSuccess()) {
                return@runCatching FetchResult(false, "", "取不到（HTTP ${response.status.value}）")
            }
            val body = response.bodyAsText()
            FetchResult(true, extractText(body))
        }.getOrElse { error ->
            FetchResult(false, "", "取不到：${error.message ?: error::class.simpleName}")
        }
    }

    /** 主机名是 IP 字面量且在私有段里，或者直接写 localhost。 */
    private fun looksInternal(url: String): Boolean {
        val host = url.removePrefix("https://").substringBefore('/').substringBefore(':').lowercase()
        if (host == "localhost" || host.endsWith(".local")) return true

        val parts = host.split('.').mapNotNull { it.toIntOrNull() }
        if (parts.size != 4) return false
        return when {
            parts[0] == 10 -> true
            parts[0] == 127 -> true
            parts[0] == 192 && parts[1] == 168 -> true
            parts[0] == 172 && parts[1] in 16..31 -> true
            parts[0] == 169 && parts[1] == 254 -> true
            else -> false
        }
    }

    /**
     * 最朴素的正文提取：丢掉 script / style 的**内容**，再去掉所有标签，
     * 折叠空白，最后截断。
     *
     * 不引 HTML 解析库的理由：一期只看"能不能拿到大致可读的文字"，
     * 正则够用。真正做差了再换，换的时候只动这一个函数。
     */
    private fun extractText(html: String): String {
        val withoutBlocks = html
            .replace(Regex("(?is)<script[^>]*>.*?</script>"), " ")
            .replace(Regex("(?is)<style[^>]*>.*?</style>"), " ")
            .replace(Regex("(?is)<!--.*?-->"), " ")
        val withoutTags = withoutBlocks.replace(Regex("(?s)<[^>]+>"), " ")
        val decoded = withoutTags
            .replace("&nbsp;", " ")
            .replace("&amp;", "&")
            .replace("&lt;", "<")
            .replace("&gt;", ">")
            .replace("&quot;", "\"")
            .replace("&#39;", "'")
        return decoded
            .replace(Regex("\\s+"), " ")
            .trim()
            .take(MAX_CHARS)
    }

    private companion object {
        /** 单次网页正文的上限。够模型读懂，又不会把上下文撑爆。 */
        const val MAX_CHARS = 8000
    }
}

```kotlin
    /**
     * 按关键词检索记忆。给 recall_memory 工具用。
     *
     * 一期没有向量检索——记忆条数本来就少，关键词匹配够用。
     * 参数是 query 而不是 id，是为了将来换成向量检索时接口不用改。
     */
    fun recallMemories(agentId: String, query: String, limit: Int): List<MemoryItem> {
        val pool = memoriesOf(agentId)
        val needle = query.trim().lowercase()
        if (needle.isEmpty()) return pool.take(limit)

        return pool
            .map { it to scoreMemory(it, needle) }
            .filter { it.second > 0 }
            .sortedByDescending { it.second }
            .take(limit)
            .map { it.first }
    }

    /** 命中字数。粗糙但够用，而且结果稳定可测。 */
    private fun scoreMemory(item: MemoryItem, needle: String): Int {
        val haystack = item.text.lowercase()
        return needle.length.let { _ ->
            // 整串命中给高分，否则按逐字命中计数
            if (haystack.contains(needle)) 100 + needle.length
            else needle.count { it in haystack }
        }
    }

    /**
     * 从工具建任务。跟 [confirmDraft] 不同的是：**这里不弹确认卡片，直接落库**。
     *
     * 为什么允许：模型的判断由工具描述约束（"只在用户明确要托管时调用"），
     * 而且工具调用对用户是可见的（执行卡片）。草稿卡片那条路保留给文本降级。
     */
    fun createTaskFromTool(
        agentId: String,
        title: String,
        recurrence: Recurrence,
        scheduleLabel: String,
        note: String?,
    ): String {
        val id = nextId("task")
        val task = scheduled(
            AgentTask(
                id = id,
                agentId = agentId,
                title = title,
                recurrence = recurrence,
                scheduleLabel = scheduleLabel,
                nextRunLabel = "",
                note = note,
            )
        )
        tasks.add(0, task)
        persistState()
        return id
    }

    /** 给模型看的任务摘要，不带 nextRunAt 这种调度字段。 */
    fun taskSummariesFor(agentId: String): List<TaskSummary> =
        tasks.filter { it.agentId == agentId }
            .filter { it.status != TaskStatus.Done }
            .map {
                TaskSummary(
                    id = it.id,
                    title = it.title,
                    recurrence = it.recurrence,
                    scheduleLabel = it.scheduleLabel,
                    nextRunLabel = it.nextRunLabel,
                    status = it.status,
                )
            }

    /** 改任务。**核对 agentId**——模型不该能改到别的智能体的任务。 */
    fun updateTaskFromTool(
        agentId: String,
        id: String,
        title: String?,
        scheduleLabel: String?,
    ): Boolean {
        val index = tasks.indexOfFirst { it.id == id && it.agentId == agentId }
        if (index < 0) return false

        val current = tasks[index]
        val updated = current.copy(
            title = title?.takeIf { it.isNotBlank() } ?: current.title,
            scheduleLabel = scheduleLabel?.takeIf { it.isNotBlank() } ?: current.scheduleLabel,
        )
        // 改了时间就要重排期，否则它还会按旧时刻跑
        tasks[index] = if (scheduleLabel.isNullOrBlank()) updated else scheduled(updated)
        persistState()
        return true
    }

    /** 取消任务。**置为 Done 而不是删掉**——删了用户就看不到它出过什么。 */
    fun cancelTaskFromTool(agentId: String, id: String): Boolean {
        val index = tasks.indexOfFirst { it.id == id && it.agentId == agentId }
        if (index < 0) return false
        tasks[index] = tasks[index].copy(status = TaskStatus.Done)
        persistState()
        return true
    }

import com.starringbox.staragent.core.model.MemoryCategory

/**
 * 把关于用户的一条信息存进长期记忆。
 *
 * 描述里**明确说了什么时候不要调**——这是工具描述里最要紧的一句。
 * 不写的话模型会把每轮闲聊都记下来，记忆很快就被噪音塞满，
 * 而且用户会开始怀疑"它到底记住了些什么"。
 */
object RememberTool : Tool {

    override val definition = ToolDefinition(
        name = "remember",
        description = "把关于用户的一条长期信息存下来，让他下次不用重复说。" +
            "只在用户**明确**表达了喜好、身边重要的人、生活习惯、经历或决策时调用。" +
            "**不要把闲聊、问候、以及临时性的请求记下来**——那样记忆会被噪音淹没。" +
            "同一件事已经记过就不要重复记。",
        inputSchema = buildJsonObject {
            put("type", "object")
            putJsonObject("properties") {
                putJsonObject("category") {
                    put("type", "string")
                    put("description", "这条信息属于哪一类")
                    put("enum", buildJsonArray {
                        MemoryCategory.entries.forEach { add(it.name) }
                    })
                }
                putJsonObject("text") {
                    put("type", "string")
                    put("description", "要记住的这件事，写成一句完整的话")
                }
            }
            putJsonArray("required") {
                add("category")
                add("text")
            }
        },
    )

    override suspend fun execute(args: JsonObject, host: ToolHost): ToolResult {
        val rawCategory = args.stringOrNull("category")
            ?: return ToolResult(callId = "", content = "缺参数 category", isError = true)

        val category = MemoryCategory.entries.firstOrNull { it.name == rawCategory }
            ?: return ToolResult(
                callId = "",
                content = "category 只能是这些之一：${MemoryCategory.entries.joinToString("、") { it.name }}",
                isError = true,
            )

        val text = args.stringOrNull("text")
        if (text.isNullOrBlank()) {
            return ToolResult(callId = "", content = "text 不能为空", isError = true)
        }

        host.memory.remember(host.agentId, category, text.trim())
        return ToolResult(callId = "", content = "已记住：$text")
    }
}

/**
 * 按关键词检索当前智能体的记忆。
 *
 * 这是这一批里价值最高的一个：它把记忆从"静态列表"变成"参与推理"。
 * 在此之前记忆只能靠系统提示灌最近若干条，模型没法主动按需查。
 */
object RecallMemoryTool : Tool {

    override val definition = ToolDefinition(
        name = "recall_memory",
        description = "从你记得的关于这位用户的事里查找相关条目。" +
            "当用户提到过去说过的事、你拿不准他偏好什么、或者要做跟他的习惯相关的判断时，先查一下。" +
            "不要为了每条消息都查一次。",
        inputSchema = buildJsonObject {
            put("type", "object")
            putJsonObject("properties") {
                putJsonObject("query") {
                    put("type", "string")
                    put("description", "要查的关键词或短句")
                }
            }
            putJsonArray("required") { add("query") }
        },
    )

    override suspend fun execute(args: JsonObject, host: ToolHost): ToolResult {
        val query = args.stringOrNull("query").orEmpty()
        if (query.isBlank()) {
            return ToolResult(callId = "", content = "query 不能为空", isError = true)
        }

        val hits = host.memory.recall(host.agentId, query, limit = DEFAULT_LIMIT)
        if (hits.isEmpty()) {
            // 明确说"没有"，否则模型会以为工具坏了、反复重试
            return ToolResult(callId = "", content = "没有查到跟「$query」相关的记忆。")
        }

        val text = hits.joinToString("\n") { "- ${it.category.label}：${it.text}" }
        return ToolResult(callId = "", content = "查到 ${hits.size} 条：\n$text")
    }

    private const val DEFAULT_LIMIT = 10
}

/** 从 JSON 对象里取字符串。缺字段或类型不对都返回 null。 */
internal fun JsonObject.stringOrNull(key: String): String? {
    val value = this[key] ?: return null
    return (value as? kotlinx.serialization.json.JsonPrimitive)?.content
}

 *
 * 每一"轮"是一个事件列表。脚本用完就当作不再有工具调用（自然收尾）。

private fun scriptedStream(
    rounds: List<List<ChatStreamEvent>>,
): suspend (LlmEndpoint, List<LlmTurn>) -> Flow<ChatStreamEvent> {
    var index = 0
    return { _, _ ->
        flow {
            val events = rounds.getOrNull(index) ?: emptyList()
            index++
            events.forEach { emit(it) }
            emit(ChatStreamEvent.Completed)
        }
    }
}

package com.starringbox.staragent.core.data

import com.starringbox.staragent.core.llm.ChatStreamEvent
import com.starringbox.staragent.core.llm.LlmEndpoint
import com.starringbox.staragent.core.llm.LlmRole
import com.starringbox.staragent.core.llm.LlmTurn
import com.starringbox.staragent.core.model.ToolCall
import com.starringbox.staragent.core.model.ToolResult
import com.starringbox.staragent.core.tools.ToolHost
import com.starringbox.staragent.core.tools.ToolRegistry
import kotlinx.coroutines.flow.Flow
import kotlinx.serialization.json.Json
import kotlinx.serialization.json.JsonObject
import kotlinx.serialization.json.jsonObject

/** 循环的结局。 */
internal sealed interface LoopOutcome {
    /** 模型不再要工具了，[text] 是它最后说的那段话。 */
    data class Done(val text: String) : LoopOutcome

    /** 撞上步数上限。[reason] 是给用户看的话。 */
    data class Stopped(val reason: String) : LoopOutcome

    data class Failed(val message: String) : LoopOutcome
}

/**
 * 一次多步执行：请求 → 调工具 → 回喂 → 再请求，直到模型不再要工具。
 *
 * [stream] 是**注入**的，不是直接调 ChatCompletion——这样循环能用假 LLM 单测，
 * 不用起 HTTP。这是它能被测的唯一前提。
 *
 * 工具是**串行**执行的：每个工具都可能是一次网络往返，串行让成本可预测，
 * 也跟现有"群里成员依次发言"的形状一致。并行是后续的优化。
 */
internal class AgentLoop(
    private val stream: suspend (LlmEndpoint, List<LlmTurn>) -> Flow<ChatStreamEvent>,
    private val maxIterations: Int = DEFAULT_MAX_ITERATIONS,
) {

    suspend fun run(
        endpoint: LlmEndpoint,
        history: MutableList<LlmTurn>,
        host: ToolHost,
        registry: ToolRegistry,
        onText: (String) -> Unit,
        onToolStart: (ToolCall) -> Unit,
        onToolEnd: (ToolCall, ToolResult) -> Unit,
    ): LoopOutcome {
        var iteration = 0

        while (true) {
            iteration++
            if (iteration > maxIterations) {
                // 明确告诉用户，不要静默截断——他得知道为什么停在这
                return LoopOutcome.Stopped(
                    "这个任务需要的步骤有点多，我先停在这里。你可以把要求说得更具体一些。"
                )
            }

            val buffer = StringBuilder()
            var calls: List<ToolCall> = emptyList()
            var failure: String? = null

            stream(endpoint, history.toList()).collect { event ->
                when (event) {
                    is ChatStreamEvent.Delta -> {
                        buffer.append(event.text)
                        onText(event.text)
                    }
                    is ChatStreamEvent.ToolCallsReady -> calls = event.calls
                    is ChatStreamEvent.Failed -> failure = event.message
                    ChatStreamEvent.Completed -> Unit
                }
            }

            failure?.let { return LoopOutcome.Failed(it) }

            val spoken = buffer.toString()

            // assistant 这一轮要按原样写回历史：两家回喂时都要求把带工具调用的
            // assistant 轮带上，否则模型看到自己"没说过"那些调用，会重复请求。
            history += LlmTurn(
                role = LlmRole.Assistant,
                text = spoken,
                toolCalls = calls,
            )

            if (calls.isEmpty()) {
                // 终止条件：模型不再要工具了
                return LoopOutcome.Done(spoken)
            }

            val results = calls.map { call ->
                onToolStart(call)
                val result = execute(call, host, registry)
                onToolEnd(call, result)
                result
            }

            // 工具结果作为一轮回喂。两家的**放置位置不同**由 core:llm 内部处理，
            // 这里只是把结果挂到 turn 上。
            history += LlmTurn(
                role = LlmRole.User,
                text = "",
                toolResults = results,
            )
        }
    }

    /**
     * 执行一次调用。**所有失败都变成结果，不变成异常**——
     * 工具的错误是给模型看的信息，它看到"参数格式错误"会自己重试；
     * 异常穿出去只会让用户看到崩溃。
     */
    private suspend fun execute(
        call: ToolCall,
        host: ToolHost,
        registry: ToolRegistry,
    ): ToolResult {
        val tool = registry.find(call.name)
            ?: return ToolResult(
                callId = call.id,
                content = "没有名为「${call.name}」的工具。可用的有：" +
                    registry.definitions().joinToString("、") { it.name },
                isError = true,
            )

        val args = parseArguments(call.arguments)
            ?: return ToolResult(
                callId = call.id,
                content = "参数不是合法的 JSON，请重新调用并确保参数是一个 JSON 对象。",
                isError = true,
            )

        return runCatching { tool.execute(args, host) }
            .getOrElse { error ->
                ToolResult(
                    callId = call.id,
                    content = "工具执行出错：${error.message ?: error::class.simpleName}",
                    isError = true,
                )
            }
            .copy(callId = call.id) // 工具自己不知道 callId，在这里补上
    }

    /** 参数解析失败返回 null，由调用方变成一条错误结果——不抛。 */
    private fun parseArguments(raw: String): JsonObject? =
        runCatching { Json.parseToJsonElement(raw.ifBlank { "{}" }).jsonObject }.getOrNull()

    companion object {
        /**
         * 步数上限。
         *
         * 8 是个折中：够"查资料 → 读正文 → 记住 → 回答"这种多步任务，
         * 又不至于在模型死循环时烧掉太多 token。
         */
        const val DEFAULT_MAX_ITERATIONS = 8
    }
}

 * 一期**只做 fetch，不做 search**：搜索需要一个后端（等于又一把 Key），
 * 而 fetch 是纯 HTTP，用户直接把地址给出来就行。搜索留待后端定了再说。

            "当用户给了一个链接、或者你需要看某个具体页面的内容时用。" +
            "只能打开 https 开头的地址；内网地址打不开。" +
            "**这不是搜索引擎**——你不知道地址的话，直接问用户要。",
        inputSchema = buildJsonObject {
            put("type", "object")
            putJsonObject("properties") {
                putJsonObject("url") {
                    put("type", "string")
                    put("description", "要打开的网页地址，必须 https 开头")
                }
            }
            putJsonArray("required") { add("url") }
        },
    )

    override suspend fun execute(args: JsonObject, host: ToolHost): ToolResult {
        val url = args.stringOrNull("url")?.trim()
        if (url.isNullOrBlank()) {
            return ToolResult(callId = "", content = "url 不能为空", isError = true)

        // 先在工具这层挡一道，不把明显不合格的地址发给端口——

package com.starringbox.staragent.core.data

import com.starringbox.staragent.core.model.ExecutionStep
import com.starringbox.staragent.core.model.ToolCall
import com.starringbox.staragent.core.model.ToolResult

/**
 * 把工具调用翻成用户看得懂的步骤。
 *
 * 界面上显示的必须是**在做什么**（"查资料"），不是工具的内部名字（"web_fetch"）。
 * 用户不关心我们内部把工具叫什么。
 *
 * 一期内置工具就那么几个，所以用一张表映射；表里没有的工具退回它的描述性名字，
 * 而不是空白——将来加工具忘了更新这张表，界面上也还是个能看的词。
 */
internal fun executionStepsFor(
    calls: List<ToolCall>,
    results: List<ToolResult>,
): List<ExecutionStep> {
    val doneIds = results.map { it.callId }.toSet()
    return calls.map { call ->
        ExecutionStep(
            label = readableLabelFor(call),
            done = call.id in doneIds,
        )
    }
}

private fun readableLabelFor(call: ToolCall): String = when (call.name) {
    "recall_memory" -> "回忆关于你的事"
    "remember" -> "记住这件事"
    "create_task" -> "创建托管任务"
    "list_tasks" -> "查看任务"
    "update_task" -> "调整任务"
    "cancel_task" -> "取消任务"
    "web_fetch" -> "读取网页"
    else -> call.name
}
